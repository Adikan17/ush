// Merged app: Routing + Login/Cart + Dynamic Form + Users (useEffect)
// Setup: npm install react-router-dom   |   Save as src/App.jsx
import { useState, useEffect } from "react";
import { BrowserRouter, Routes, Route, Link, Navigate, useParams, useNavigate } from "react-router-dom";

// ---------- Data ----------
const PRODUCTS = [
  { id: 1, name: "Laptop", price: 55000 },
  { id: 2, name: "Headphones", price: 2500 },
  { id: 3, name: "Backpack", price: 1200 },
];

const FIELDS = [ // the form is built from this config
  { name: "name", label: "Name", type: "text" },
  { name: "email", label: "Email", type: "email" },
  { name: "course", label: "Course", type: "select", options: ["BCA", "B.Tech", "MCA"] },
  { name: "about", label: "About", type: "textarea" },
];

const fakeFetch = () => // pretends to be an API call
  new Promise((ok) => setTimeout(() => ok([
    { id: 1, name: "Leanne", city: "Hyderabad" },
    { id: 2, name: "Ervin", city: "Chennai" },
  ]), 1500));

// ---------- Pages ----------
const Home = () => <h2>Welcome! <Link to="/products">Shop now</Link></h2>;

const Products = () => (
  <>
    <h2>Products</h2>
    {PRODUCTS.map((p) => (
      <p key={p.id}>{p.name} - ₹{p.price} <Link to={`/products/${p.id}`}>Details</Link></p>
    ))}
  </>
);

function Product({ add }) {
  const { id } = useParams(); // dynamic route
  const p = PRODUCTS.find((x) => x.id === +id);
  return p ? (
    <>
      <h2>{p.name} - ₹{p.price}</h2>
      <button onClick={() => add(p)}>Add to cart</button>
    </>
  ) : <h2>Not found</h2>;
}

function Cart({ cart }) {
  return (
    <>
      <h2>Cart</h2>
      {cart.map((p, i) => <p key={i}>{p.name} - ₹{p.price}</p>)}
      <b>Total: ₹{cart.reduce((s, p) => s + p.price, 0)}</b>
    </>
  );
}

function Login({ setUser }) {
  const [name, setName] = useState("");
  const go = useNavigate();
  return (
    <>
      <h2>Login (any name)</h2>
      <input placeholder="Name" onChange={(e) => setName(e.target.value)} />{" "}
      <button onClick={() => { if (name) { setUser(name); go("/cart"); } }}>Login</button>
    </>
  );
}

function Form() {
  const [data, setData] = useState({});
  const [done, setDone] = useState(false);
  const set = (e) => setData({ ...data, [e.target.name]: e.target.value });

  if (done) return <pre>{JSON.stringify(data, null, 2)}</pre>;
  return (
    <form onSubmit={(e) => { e.preventDefault(); setDone(true); }}>
      <h2>Form</h2>
      {FIELDS.map((f) => (
        <p key={f.name}>
          {f.label}:{" "}
          {f.type === "select" ? (
            <select name={f.name} onChange={set}>
              <option value="">--</option>
              {f.options.map((o) => <option key={o}>{o}</option>)}
            </select>
          ) : f.type === "textarea" ? (
            <textarea name={f.name} onChange={set} />
          ) : (
            <input name={f.name} type={f.type} onChange={set} />
          )}
        </p>
      ))}
      <button>Submit</button>
    </form>
  );
}

function Users() {
  const [users, setUsers] = useState(null); // null = still loading
  useEffect(() => { fakeFetch().then(setUsers); }, []);

  if (!users) return <p>Loading...</p>;
  return (
    <>
      <h2>Users</h2>
      {users.map((u) => <p key={u.id}>{u.name} - {u.city}</p>)}
    </>
  );
}

// ---------- App ----------
export default function App() {
  const [user, setUser] = useState(null);
  const [cart, setCart] = useState([]);

  return (
    <BrowserRouter>
      <nav style={{ background: "#eee", padding: 10, display: "flex", gap: 15 }}>
        <Link to="/">Home</Link>
        <Link to="/products">Products</Link>
        <Link to="/cart">Cart ({cart.length})</Link>
        <Link to="/form">Form</Link>
        <Link to="/users">Users</Link>
        {user ? <span>Hi, {user}</span> : <Link to="/login">Login</Link>}
      </nav>
      <div style={{ padding: 20 }}>
        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/products" element={<Products />} />
          <Route path="/products/:id" element={<Product add={(p) => setCart([...cart, p])} />} />
          <Route path="/cart" element={user ? <Cart cart={cart} /> : <Navigate to="/login" />} />
          <Route path="/login" element={<Login setUser={setUser} />} />
          <Route path="/form" element={<Form />} />
          <Route path="/users" element={<Users />} />
          <Route path="*" element={<h2>404 - Not Found</h2>} />
        </Routes>
      </div>
    </BrowserRouter>
  );
}
